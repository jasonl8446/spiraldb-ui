// final-verify D196: what Origin the real browser sends on a UI write from the stack's own page, and the answer it gets.

const b = await chromium.launch(); const p = await b.newPage();
p.on('requestfinished', async (r) => { if (r.url().includes('/api/') && r.method() !== 'GET') { const h = await r.allHeaders(); console.log(`[browser] ${r.method()} ${new URL(r.url()).pathname} Origin=${h.origin ?? '(none)'} Sec-Fetch-Site=${h['sec-fetch-site'] ?? '(none)'} (allHeaders: the headers as sent on the wire)`); } });
p.on('response', (r) => { if (r.url().includes('/api/') && r.request().method() !== 'GET') console.log(`[browser]   -> ${r.status()}`); });
await p.goto('http://localhost:3292/quests/WC-UNICORN-MAIN-007');
await p.getByRole('button', { name: 'Mark Verified' }).click();   // a UI write: PATCH /api/status/quests/WC-UNICORN-MAIN-007
