// final-verify: per clone quest, how many dialog entries the D195 rule opens the 'Advanced' accordion for (value only; errors aside).
import fs from 'node:fs';
import JSON5 from 'json5';
import { DIALOG_ACCORDIONS, dialogFieldsInAccordion } from '../../client/src/lib/quest-dialog';
import { isDefaultFieldValue } from '../../client/src/lib/advanced';
const adv = DIALOG_ACCORDIONS.find((a) => a.id === 'Advanced')!;
const keys = dialogFieldsInAccordion(adv).map((f) => f.key);
const dir = 'data/test-spiraldb/QuestTemplates';
let total = 0, opened = 0; const mixed: string[] = [];
for (const f of fs.readdirSync(dir).filter((x) => x.endsWith('.json')).sort()) {
  let q: any; try { q = JSON5.parse(fs.readFileSync(`${dir}/${f}`, 'utf8')); } catch { continue; }
  const entries: any[] = [];
  const walk = (l: any) => (l?.m_dialogs ?? []).forEach((g: any) => (g.m_dialogEntries ?? []).forEach((e: any) => entries.push(e)));
  walk(q.m_dialogList); (q.m_goals ?? []).forEach((g: any) => walk(g.m_dialogList));
  const o = entries.filter((e) => keys.some((k) => !isDefaultFieldValue(k, e[k], e.$type))).length;
  total += entries.length; opened += o;
  if (o > 0 && o < entries.length) mixed.push(`${q.m_questName}:${o}/${entries.length}`);
  if (q.m_questName === 'WC-UNICORN-MAIN-001') console.log('WC-UNICORN-MAIN-001 Advanced opened by value:', o, 'of', entries.length);
}
console.log('clone (all 322 QuestTemplates, JSON5-parsed): entries', total, 'Advanced opened by value', opened);
console.log('quests with both a collapsed and an opened Advanced (first 8):', mixed.slice(0, 8).join(' '), '... total', mixed.length);
