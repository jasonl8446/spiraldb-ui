const ensureEdit = async (page) => {
  const root = page.getByRole('main').locator('[data-edit-mode]').first();
  if ((await root.count()) && (await root.getAttribute('data-edit-mode')) === 'true') return;
  await editBtn(page).click();

        const m = await page.evaluate(() => ({
          over: document.documentElement.scrollWidth - document.documentElement.clientWidth,

          burger: !!document.querySelector('[aria-label="Open navigation"]'),
        }));
        cells++;
        rails.add(`${m.rail}`);

      }
      log(A, `width ${w}: sidebar widths seen ${JSON.stringify([...rails])}`);
    }
    log(A, `route x width cells: ${cells}; overflow violations (>1px): ${bad.length ? JSON.stringify(bad) : 'none'}`);

const browser = await chromium.launch();
let failed = 0;
try {
  for (const arm of process.argv.slice(2)) {
    const ctx = await browser.newContext({ viewport: { width: 1440, height: 1000 } });
    const page = await ctx.newPage();
    const errors = [];
    page.on('console', (m) => m.type() === 'error' && errors.push(flat(m.text()).slice(0, 160)));
    try {
      await arms[arm](page, browser);
      console.log(`[${arm}] ARM DONE`);
    } catch (e) {
      failed++;
      console.log(`[${arm}] ARM FAILED: ${flat(String(e)).slice(0, 600)}`);
      if (SHOTS) await page.screenshot({ path: path.join(SHOTS, `FAILED-${arm}.png`) }).catch(() => {});
    }
    if (errors.length) console.log(`[${arm}] console errors: ${JSON.stringify([...new Set(errors)].slice(0, 5))}`);
    await ctx.close();
  }
} finally {
  await browser.close();
}
fs.writeSync(1, '');
process.exit(failed ? 1 : 0);
