# d45diff.mjs
// D45 before/after: every leaf path whose value differs between two wrapper outputs.
import fs from 'node:fs';
const [a, b] = process.argv.slice(2).map((f) => JSON.parse(fs.readFileSync(f, 'utf8')));
const out = [];
const walk = (x, y, p) => {
  if (x !== null && y !== null && typeof x === 'object' && typeof y === 'object') {
    for (const k of new Set([...Object.keys(x), ...Object.keys(y)])) {
      const name = Array.isArray(x) && x[k]?.m_goalName ? `[${x[k].m_goalName}]` : Array.isArray(x) ? `[${k}]` : `.${k}`;
      walk(x[k], y[k], p + name);
    }
  } else if (JSON.stringify(x) !== JSON.stringify(y)) out.push(`${p}: ${JSON.stringify(x)} -> ${JSON.stringify(y)}`);
};
walk(a, b, '');
console.log(out.length ? out.join('\n') : '(identical)');
