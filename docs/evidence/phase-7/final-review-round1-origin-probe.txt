import { chromium } from '@playwright/test';
const browser = await chromium.launch();
const page = await browser.newPage();
const seen = [];
for (const base of process.argv.slice(2)) {
  await page.goto(`${base}/`);
  const result = await page.evaluate(async () => {
    const r = await fetch('/api/suggestions/999999/reject', { method: 'POST' });
    return { status: r.status, body: await r.text() };
  });
  seen.push(`${base} same-origin POST /api/suggestions/999999/reject -> ${result.status} ${result.body}`);
}
// A foreign page: about:blank has an opaque origin -> Origin: null on a no-cors POST.
await page.goto('about:blank');
const target = process.argv[2];
await page.setContent('<p>foreign</p>');
const foreign = await page.evaluate(async (url) => {
  try { await fetch(`${url}/api/suggestions/999999/reject`, { method: 'POST', mode: 'no-cors' }); return 'sent'; } catch (e) { return String(e); }
}, target);
seen.push(`about:blank no-cors POST -> ${foreign} (status unreadable to the page; server log shows the answer)`);
console.log(seen.join('\n'));
await browser.close();
