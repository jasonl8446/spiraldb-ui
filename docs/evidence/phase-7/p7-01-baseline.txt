// p7-01 baseline re-measure. Usage: node baseline.mjs <corpus-root> <label>
// Reads every QuestTemplates/*.json with JSON5 (the corpus carries trailing commas, D48(b)).
import fs from 'node:fs';
import path from 'node:path';
import { createRequire } from 'node:module';
const require = createRequire('/home/jason/Documents/git-projects/spiraldb-ui/package.json');
const JSON5 = require('json5');

const [root, label] = process.argv.slice(2);
const dir = path.join(root, 'QuestTemplates');
const files = fs.readdirSync(dir).filter((f) => f.endsWith('.json')).sort();
const nonJson = fs.readdirSync(dir).filter((f) => !f.endsWith('.json'));
const empty = (v) =>
  v === undefined || v === null || v === '' || (Array.isArray(v) && v.length === 0);
const short = (t) => (typeof t === 'string' ? t.split(',')[0].split('.').pop() : String(t));

let quests = 0, parseErr = 0;
let goalPatternAll = 0, goalPatternAny = 0, noGoals = 0;
let fourTextEmpty = 0;
let nameIdZero = 0, activityDefault = 0, bothIdAct = 0;
const activity = {};
let goals = 0;
const goalClass = {};
let completeTextEmpty = 0, hyperlinkEmpty = 0, goalReqEmpty = 0;
let endResultsEmpty = 0;
const resultClass = {};
const questsWithResultClass = {};
for (const f of files) {
  let q;
  try { q = JSON5.parse(fs.readFileSync(path.join(dir, f), 'utf8')); } catch { parseErr++; continue; }
  quests++;
  const gs = Array.isArray(q.m_goals) ? q.m_goals : [];
  if (gs.length === 0) noGoals++;
  const matches = gs.map((g) => /^\d+_/.test(String(g.m_goalName ?? '')));
  if (gs.length > 0 && matches.every(Boolean)) goalPatternAll++;
  if (matches.some(Boolean)) goalPatternAny++;
  if (['m_questInfo', 'm_questPrep', 'm_questUnderway', 'm_questComplete'].every((k) => empty(q[k]))) fourTextEmpty++;
  const idZero = (q.m_questNameID ?? 0) === 0;
  const actDef = q.m_activityType === undefined || q.m_activityType === 'ACTIVITY_NotActivity' || q.m_activityType === 0;
  activity[String(q.m_activityType)] = (activity[String(q.m_activityType)] ?? 0) + 1;
  if (idZero) nameIdZero++;
  if (actDef) activityDefault++;
  if (idZero && actDef) bothIdAct++;
  for (const g of gs) {
    goals++;
    const c = short(g.$type);
    goalClass[c] = (goalClass[c] ?? 0) + 1;
    if (empty(g.m_completeText)) completeTextEmpty++;
    if (empty(g.m_hyperlink)) hyperlinkEmpty++;
    const r = g.m_goalRequirements;
    if (empty(r) || empty(r.m_requirements)) goalReqEmpty++;
  }
  const er = q.m_endResults?.m_results;
  if (empty(er)) endResultsEmpty++;
  const seen = new Set();
  for (const r of er ?? []) {
    const c = short(r.$type);
    resultClass[c] = (resultClass[c] ?? 0) + 1;
    seen.add(c);
  }
  for (const c of seen) questsWithResultClass[c] = (questsWithResultClass[c] ?? 0) + 1;
}
console.log(JSON.stringify({
  corpus: label, root, quest_files: files.length, non_json_entries: nonJson, parsed: quests, parse_errors: parseErr,
  goal_name_pattern: { every_goal_matches: goalPatternAll, any_goal_matches: goalPatternAny, files_with_no_goals: noGoals },
  quest_info_prep_underway_complete_all_empty: fourTextEmpty,
  questNameID_zero: nameIdZero, activityType_default: activityDefault, both: bothIdAct, activityType_values: activity,
  goals, goal_classes: goalClass,
  goal_completeText_empty: completeTextEmpty, goal_hyperlink_empty: hyperlinkEmpty, goal_goalRequirements_empty: goalReqEmpty,
  endResults_empty: endResultsEmpty, endResults_entries_by_class: resultClass, quests_carrying_class: questsWithResultClass,
}, null, 1));
